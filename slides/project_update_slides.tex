\begin{frame}[t]{Final Project Proposal: Clinical Problem \& Pipeline}
\scriptsize
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{block}{Clinical problem}
\begin{itemize}
  \item ER/PR/HER2 status decides breast-cancer therapy; clinically it is measured by \textbf{IHC}.
  \item IHC disagrees with mRNA-level status in a sizeable minority; for HER2 0 vs 1+ (HER2-low, T-DXd eligibility) pathologists agree in only $\sim$26\% of cases.
  \item H\&E AI models learn whichever label a dataset provides: \textbf{nobody knows which biological truth they track}.
\end{itemize}
\end{block}
\begin{alertblock}{Research question}
When IHC, RNA and PAM50 disagree, which one does an H\&E foundation-model classifier follow? Does it change between the FFPE diagnostic slide and the frozen slide cut from the sequenced tissue?
\end{alertblock}
\end{column}
\begin{column}{0.49\textwidth}
\begin{block}{Five standalone experiments (stricter evaluation, no new architecture)}
\begin{itemize}
  \item[\textbf{1a}] IHC- vs RNA-defined label: which is more predictable from H\&E?
  \item[\textbf{1b}] Train on concordant cases; on IHC$\neq$RNA cases, which side does the model take?
  \item[\textbf{1c}] FFPE-DX vs frozen slide $\times$ IHC vs RNA label (interaction)
  \item[\textbf{1d}] HER2 0 vs 1+: separable from H\&E vs from ERBB2 mRNA?
  \item[\textbf{1e}] Within-patient agreement: top (TS) vs bottom (BS) frozen slide
\end{itemize}
\end{block}
\end{column}
\end{columns}

\begin{frame}[t]{Baseline Experiments \& Analysis}
\scriptsize
\textbf{Setup:} precomputed TITAN slide embeddings (TRACE, 1,125 DX slides) $\rightarrow$ patient mean $\rightarrow$ L2 logistic regression; 5-fold CV, \emph{random} vs \emph{site-grouped} (40 sites); 95\% bootstrap CIs. Laptop CPU, $\sim$10 min. (Linear probes; final models use ABMIL on patch features.)
\vspace{2pt}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
\centering
\begin{tabular}{lrrr}
\toprule
H\&E $\rightarrow$ label & $n$ & AUC random & AUC site \\
\midrule
ER (IHC) & 1,004 & 0.893 & 0.885 \\
ER (RNA) & 1,039 & 0.936 & 0.921 \\
PR (IHC) & 1,001 & 0.804 & 0.792 \\
PR (RNA) & 1,039 & 0.797 & 0.787 \\
HER2 (clinical) & 590 & 0.733 & 0.674 \\
HER2 (ERBB2 amp) & 1,027 & 0.757 & 0.755 \\
HER2 0 vs 1+ & 318 & 0.633 & \textbf{0.433} \\
Shuffled ER (control) & 1,004 & 0.431 & 0.460 \\
\bottomrule
\end{tabular}
\raggedright
\begin{block}{Sanity / leakage checks}
\begin{itemize}
  \item Shuffled labels $\rightarrow$ AUC $\approx$ 0.46 \checkmark{} (no CV leakage)
  \item Site-only model (no image): AUC 0.53--0.59
  \item Embedding identifies the site: 81.6\% vs 18.4\% chance $\rightarrow$ site-grouped CV is mandatory
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.52\textwidth}
\begin{block}{Preliminary 1a: IHC vs RNA label (same patients and folds)}
\begin{itemize}
  \item ER ($n$=989): IHC 0.882 vs RNA 0.930, $\Delta$ = \textbf{+0.048} [95\% CI +0.023, +0.074]
  \item PR ($n$=986): $\Delta$ = +0.009 [$-$0.022, +0.041]
  \item HER2 ($n$=568): $\Delta$ = +0.047 [$-$0.015, +0.104]
\end{itemize}
\end{block}
\begin{block}{Preliminary 1b: train on ER-concordant, score 54 discordant}
\begin{itemize}
  \item Model sides with \textbf{RNA in 39/54} (binomial $p$ = 0.0015)
  \item Robust: 36/45 after dropping cases within 0.5 log$_2$ of the RNA cut-off ($p = 6.6\times10^{-5}$)
\end{itemize}
\end{block}
\begin{alertblock}{Takeaways}
\begin{itemize}
  \item For ER, H\&E morphology tracks \textbf{mRNA} more closely than clinical IHC.
  \item The apparent HER2-low signal (0.63) falls to chance under site-grouped CV: \textbf{site leakage}.
\end{itemize}
\end{alertblock}
\end{column}
